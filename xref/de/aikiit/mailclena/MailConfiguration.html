18  package de.aikiit.mailclena;
19  
20  import lombok.Builder;
21  
22  import java.util.Objects;
23  
24  /**
25   * Immutable configuration of this application.
26   *
27   * @param host mail server hostname.
28   * @param username mailbox username.
29   * @param password mailbox password, excluded from {@link #toString()}.
30   * @param command command to execute.
31   */
32  @Builder
33  public record MailConfiguration(String host, String username, String password, String command) {
34  
35      /**
36       * Creates a configuration with all required values.
37       *
38       * @param host mail server hostname.
39       * @param username mailbox username.
40       * @param password mailbox password.
41       * @param command command to execute.
42       * @throws NullPointerException if any value is null.
43       */
44      public MailConfiguration {
45          Objects.requireNonNull(host, "host");
46          Objects.requireNonNull(username, "username");
47          Objects.requireNonNull(password, "password");
48          Objects.requireNonNull(command, "command");
49      }
50  
51      @Override
52      public String toString() {
53          return "MailConfiguration[host=" + host + ", username=" + username + ", command=" + command + "]";
54      }
55  }
