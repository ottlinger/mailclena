17  package de.aikiit.mailclena;
18  
19  import org.junit.jupiter.api.Test;
20  
21  import java.util.UUID;
22  
23  import static org.assertj.core.api.Assertions.assertThat;
24  import static org.assertj.core.api.Assertions.assertThatNullPointerException;
25  
26  class MailConfigurationTest {
27  
28      @Test
29      void ensurePasswordNotInToString() {
30          String password = UUID.randomUUID().toString();
31          assertThat(MailConfiguration.builder().password(password).username("username").host("host").command("cmd").build().toString()).doesNotContain(password);
32      }
33  
34      @Test
35      void regularObjectCreation() {
36          final MailConfiguration configuration = new MailConfiguration("host", "username", "password", "myCommand");
37          assertThat(configuration.host()).isEqualTo("host");
38          assertThat(configuration.username()).isEqualTo("username");
39          assertThat(configuration.password()).isEqualTo("password");
40          assertThat(configuration.command()).isEqualTo("myCommand");
41      }
42  
43      @Test
44      void builderCreatesEquivalentRecord() {
45          final MailConfiguration configuration = MailConfiguration.builder().host("h").username("u").password("p").command("c").build();
46          assertThat(configuration.host()).isEqualTo("h");
47          assertThat(configuration.username()).isEqualTo("u");
48          assertThat(configuration.password()).isEqualTo("p");
49          assertThat(configuration.command()).isEqualTo("c");
50          assertThat(configuration).isEqualTo(new MailConfiguration("h", "u", "p", "c"));
51          assertThat(configuration.hashCode()).isEqualTo(new MailConfiguration("h", "u", "p", "c").hashCode());
52          assertThat(configuration).isNotEqualTo(new MailConfiguration("h", "u", "different", "c"));
53      }
54  
55      @Test
56      void toStringIncludesOnlyNonSecretConfiguration() {
57          assertThat(new MailConfiguration("host", "username", "secret", "list").toString())
58                  .isEqualTo("MailConfiguration[host=host, username=username, command=list]");
59      }
60  
61      @Test
62      void constructorRejectsNullValues() {
63          assertThatNullPointerException().isThrownBy(() -> new MailConfiguration(null, "u", "p", "c"))
64                  .withMessage("host");
65          assertThatNullPointerException().isThrownBy(() -> new MailConfiguration("h", null, "p", "c"))
66                  .withMessage("username");
67          assertThatNullPointerException().isThrownBy(() -> new MailConfiguration("h", "u", null, "c"))
68                  .withMessage("password");
69          assertThatNullPointerException().isThrownBy(() -> new MailConfiguration("h", "u", "p", null))
70                  .withMessage("command");
71      }
72  
73      @Test
74      void builderRejectsMissingValues() {
75          assertThatNullPointerException()
76                  .isThrownBy(() -> MailConfiguration.builder().username("u").password("p").command("c").build())
77                  .withMessage("host");
78          assertThatNullPointerException()
79                  .isThrownBy(() -> MailConfiguration.builder().host("h").password("p").command("c").build())
80                  .withMessage("username");
81          assertThatNullPointerException()
82                  .isThrownBy(() -> MailConfiguration.builder().host("h").username("u").command("c").build())
83                  .withMessage("password");
84          assertThatNullPointerException()
85                  .isThrownBy(() -> MailConfiguration.builder().host("h").username("u").password("p").build())
86                  .withMessage("command");
87      }
88  }
