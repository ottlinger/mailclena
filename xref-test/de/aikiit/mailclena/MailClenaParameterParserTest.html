44  class MailClenaParameterParserTest {
45  
46      private MailClenaParameterParser parser;
47  
48      @BeforeEach
49      void getAFreshOne() {
50          parser = new MailClenaParameterParser();
51      }
52  
53      @Test
54      void callMainWithHostnameParameter() {
55          assertThat(parser.extractConfiguration("-h=boo.foo.bar")).isEmpty();
56      }
57  
58      @Test
59      void callMainWithUsernameParameter() {
60          assertThat(parser.extractConfiguration("-u=foo")).isEmpty();
61      }
62  
63      @Test
64      void callMainWithPasswordParameter() {
65          assertThat(parser.extractConfiguration("-p=bar")).isEmpty();
66      }
67  
68      @Test
69      void callMainWithTwoParametersOneMissing() {
70          assertThat(parser.extractConfiguration("-h=bar", "-u=usr")).isEmpty();
71      }
72  
73      @Test
74      void callWithNullOrEmptyParameter() {
75          assertHelpOutput(() -> assertThat(parser.extractConfiguration((String[]) null)).isEmpty());
76          assertHelpOutput(() -> assertThat(parser.extractConfiguration()).isEmpty());
77      }
78  
79      @Test
80      void callWithoutParameterValuesButAllKeys() {
81          assertHelpOutput(() -> assertThatThrownBy(() -> parser.extractConfiguration("-h", "-u", "-p"))
82                  .isInstanceOf(IllegalArgumentException.class));
83      }
84  
85      @Test
86      void helpOutputFailureRetainsCause() throws IOException {
87          final HelpFormatter formatter = mock(HelpFormatter.class);
88          final HelpFormatter.Builder builder = mock(HelpFormatter.Builder.class);
89          final IOException failure = new IOException("Output unavailable");
90          when(builder.setShowSince(false)).thenReturn(builder);
91          when(builder.get()).thenReturn(formatter);
92          doThrow(failure).when(formatter).printHelp(eq("MailClena"), isNull(), any(Options.class), isNull(), eq(false));
93  
94          try (MockedStatic<HelpFormatter> formatters = mockStatic(HelpFormatter.class)) {
95              formatters.when(HelpFormatter::builder).thenReturn(builder);
96              assertThatThrownBy(() -> parser.extractConfiguration())
97                      .isInstanceOf(UncheckedIOException.class)
98                      .hasMessage("Unable to print command line help")
99                      .hasCause(failure);
100         }
101     }
102 
103     private void assertHelpOutput(Runnable invocation) {
104         final PrintStream originalOut = System.out;
105         final ByteArrayOutputStream output = new ByteArrayOutputStream();
106         try (PrintStream capturedOut = new PrintStream(output, true, StandardCharsets.UTF_8)) {
107             System.setOut(capturedOut);
108             invocation.run();
109         } finally {
110             System.setOut(originalOut);
111         }
112         assertThat(output.toString(StandardCharsets.UTF_8))
113                 .contains("MailClena", "--host", "--username", "--password", "--command",
114                         "Hostname", "Username", "Password", "Command to execute")
115                 .doesNotContain("Since");
116     }
117 
118     @Test
119     void callMainExtractParametersSuccessfully() {
120         final Optional<MailConfiguration> mailConfiguration = parser.extractConfiguration("-h=boo.foo.bar", "-u=foo", "-p=bar");
121         assertThat(mailConfiguration).isPresent();
122 
123         final MailConfiguration configuration = mailConfiguration.get();
124         assertThat(configuration.password()).isEqualTo("bar");
125         assertThat(configuration.username()).isEqualTo("foo");
126         assertThat(configuration.host()).isEqualTo("boo.foo.bar");
127     }
128 
129     @Test
130     void ensureAllOptionsAreMentionedAndParsed() {
131         assertThat(new MailClenaParameterParser().getAvailableOptions().getOptions()).isNotEmpty().hasSize(MailClenaParameterParser.MailClenaCommandLineOptions.values().length);
132     }
133 
134     @Test
135     void verifyCommandIsParsedIfPresent() {
136         final String unknown = "unknown";
137         final Optional<MailConfiguration> configuration = new MailClenaParameterParser().extractConfiguration("-h=host", "-u=user", "-p=password", "-c=" + unknown);
138         assertThat(configuration).isPresent();
139         assertThat(configuration.get().command()).isEqualTo(unknown);
140     }
